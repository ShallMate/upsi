cmake_minimum_required(VERSION 3.25)
project(UPSI VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/install")
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
find_package(APSI 2.0.0 EXACT CONFIG REQUIRED)
find_package(libOTe CONFIG REQUIRED COMPONENTS kkrt softspoken_ot silent_vole boost sodium)
option(UPSI_BUILD_TESTS "Build focused protocol tests" ON)
option(UPSI_ENABLE_IBLT "Build the optional IBLT PSU backend" ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(upsi_core
  aPSI.cc upsi.cc rr22/rr22.cc
  ecdhpsi/ecdh_psi.cc ecdhpsi/sender.cc ecdhpsi/receiver.cc
  psu/psu.cc psu/krtw19_psu.cc psu/krtw_psu_adapter.cc
  psu/iblt_based_psu/volePSI/SimpleIndex.cpp)
target_include_directories(upsi_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/psu/iblt_based_psu")
target_link_libraries(upsi_core PUBLIC APSI::apsi oc::libOTe OpenMP::OpenMP_CXX Threads::Threads)
target_compile_options(upsi_core PUBLIC -maes -mpclmul -mavx2)
target_compile_definitions(upsi_core PRIVATE UPSI_PSU_ENABLE_KRTW)
if(UPSI_ENABLE_IBLT)
  target_sources(upsi_core PRIVATE psu/iblt_psu.cc psu/iblt_based_psu/psu.cc psu/iblt_based_psu/iblt_h5.cc psu/iblt_based_psu/volePSI/RsOprf.cpp)
  target_compile_definitions(upsi_core PRIVATE UPSI_PSU_ENABLE_IBLT)
endif()
add_executable(upsi main.cc)
target_link_libraries(upsi PRIVATE upsi_core)
file(COPY parameters DESTINATION "${CMAKE_BINARY_DIR}/bin")
if(UPSI_BUILD_TESTS)
  enable_testing()
  add_executable(protocol_test tests/protocol_test.cc)
  target_link_libraries(protocol_test PRIVATE upsi_core)
  add_test(NAME protocol_test COMMAND protocol_test)
  set_tests_properties(protocol_test PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2" TIMEOUT 120)
  if(UPSI_ENABLE_IBLT)
    add_executable(iblt_psu_test psu/iblt_psu_smoke_test.cc)
    target_link_libraries(iblt_psu_test PRIVATE upsi_core)
    add_test(NAME iblt_psu_test COMMAND iblt_psu_test)
    set_tests_properties(iblt_psu_test PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2" TIMEOUT 120)
  endif()
endif()
